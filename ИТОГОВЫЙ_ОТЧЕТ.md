## 📝 ПРИМЕРЫ КОДА

### ✅ Правильный способ: Добавить деньги

```python
# server/database.py
def add_currency(self, character_id: int, copper: int = 0):
    """Работает одинаково для воина и мага!"""
    with self.connection() as conn:
        # Получить текущую валюту
        char = conn.execute(
            "SELECT copper FROM characters WHERE id = ?",
            (character_id,)
        ).fetchone()
        
        # Добавить и конвертировать
        total = char['copper'] + copper
        while total >= 100:
            total -= 100
            # silver += 1 ...
        
        # Сохранить в БД
        conn.execute(
            "UPDATE characters SET copper = ? WHERE id = ?",
            (total, character_id)
        )

# Использование (работает для обоих типов персонажей):
db.add_currency(warrior_id, copper=10)  # Воин получил 10 медяков
db.add_currency(mage_id, copper=10)     # Маг получил 10 медяков
```
